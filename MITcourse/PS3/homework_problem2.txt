blood_sugar = 0

is_diabetic = True
is_banana = True
is_cereal = True

anwser = input('Do you want to eat a banana?')
if anwser == 'yes':
    is_banana = True
else:
    is_banana = False

anwser = input('Do you want to eat a bowl of cereal?')
if anwser == 'yes':
    is_cereal = True
else:
    is_cereal = False

anwser = input('Are you diabetic?')
if anwser == 'yes':
    is_diabetic = True
else:
    is_diabetic = False

if is_diabetic:
    if is_banana:
        blood_sugar += 1.0
    if is_cereal:
        blood_sugar += 4.0
else:
    if is_banana:
        blood_sugar += 0.5
    if is_cereal:
        blood_sugar += 2.0

print("Estimated BSL rise:", blood_sugar,"mmol/L")
